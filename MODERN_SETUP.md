```python
COLORS = {
    'primary': '#2196F3',        # Ana mavi
    'primary_dark': '#1565C0',   # Koyu mavi (hover)
    'primary_light': '#BBDEFB',  # Açık mavi (info box)
    'success': '#4CAF50',        # Yeşil (başarı)
    'warning': '#FF9800',        # Turuncu (uyarı)
    'bg': '#F5F5F5',             # Arkaplan
    'card': '#FFFFFF',           # Kart arkaplanı
    'text': '#212121',           # Ana metin
    'text_secondary': '#757575', # İkincil metin
    'border': '#E0E0E0'          # Kenarlıklar
}
```
